import heapq
import networkx as nx
import matplotlib.pyplot as plt


# ==========================================
# Uniform Cost Search
# ==========================================

def uniform_cost_search(graph, start, goal):

    # Priority Queue
    # (total_cost, current_node)
    priority_queue = []

    heapq.heappush(priority_queue, (0, start))

    # Cost from start to each node
    cost_so_far = {
        start: 0
    }

    # Store parent of each node
    parent = {
        start: None
    }

    while priority_queue:

        # Get node with lowest cost
        current_cost, current = heapq.heappop(priority_queue)

        # Goal reached
        if current == goal:

            path = []

            while current is not None:
                path.append(current)
                current = parent[current]

            path.reverse()

            return path, current_cost

        # Explore neighbors
        for neighbor, edge_cost in graph[current]:

            new_cost = current_cost + edge_cost

            # If we found a cheaper path
            if neighbor not in cost_so_far or \
               new_cost < cost_so_far[neighbor]:

                cost_so_far[neighbor] = new_cost

                parent[neighbor] = current

                # UCS priority = g(n)
                heapq.heappush(
                    priority_queue,
                    (new_cost, neighbor)
                )

    return None, float("inf")


# ==========================================
# Graph
# ==========================================

graph = {
    'A': [('B', 2), ('C', 1)],
    'B': [('D', 3)],
    'C': [('D', 1)],
    'D': [('G', 2)],
    'G': []
}


# ==========================================
# Run UCS
# ==========================================

path, total_cost = uniform_cost_search(
    graph,
    'A',
    'G'
)

print("Uniform Cost Search")
print("-------------------")
print("Shortest Path:", " -> ".join(path))
print("Total Cost:", total_cost)


# ==========================================
# Create NetworkX Graph
# ==========================================

G = nx.DiGraph()

for node in graph:

    for neighbor, cost in graph[node]:

        G.add_edge(
            node,
            neighbor,
            weight=cost
        )


# ==========================================
# Node Positions
# ==========================================

pos = {
    'A': (0, 1),
    'B': (1, 2),
    'C': (1, 0),
    'D': (2, 1),
    'G': (3, 1)
}


# ==========================================
# Draw Nodes
# ==========================================

nx.draw_networkx_nodes(
    G,
    pos,
    node_color="lightblue",
    node_size=2000,
    edgecolors="black"
)

nx.draw_networkx_labels(
    G,
    pos,
    font_size=15,
    font_weight="bold"
)


# ==========================================
# Draw All Edges
# ==========================================

nx.draw_networkx_edges(
    G,
    pos,
    edge_color="gray",
    width=2,
    arrows=True,
    arrowsize=20
)


# ==========================================
# Display Edge Costs
# ==========================================

edge_labels = nx.get_edge_attributes(
    G,
    "weight"
)

nx.draw_networkx_edge_labels(
    G,
    pos,
    edge_labels=edge_labels,
    font_size=13,
    font_weight="bold"
)


# ==========================================
# Highlight Shortest Path
# ==========================================

path_edges = list(
    zip(path, path[1:])
)

nx.draw_networkx_edges(
    G,
    pos,
    edgelist=path_edges,
    edge_color="red",
    width=4,
    arrows=True,
    arrowsize=25
)


# ==========================================
# Display Path Cost
# ==========================================

path_costs = [
    G[u][v]["weight"]
    for u, v in path_edges
]

cost_text = " + ".join(
    str(cost) for cost in path_costs
)

plt.title(
    f"Uniform Cost Search\n"
    f"Path: {' → '.join(path)}\n"
    f"Cost: {cost_text} = {total_cost}",
    fontsize=15,
    fontweight="bold"
)

plt.axis("off")
plt.tight_layout()

plt.show()
