import heapq
import networkx as nx
import matplotlib.pyplot as plt


# -----------------------------
# Greedy Best-First Search
# -----------------------------

def greedy_best_first(graph, start, goal, heuristic):

    open_list = []

    # Greedy uses ONLY h(n)
    heapq.heappush(open_list, (heuristic[start], start))

    parent = {start: None}

    # Actual cost of the path
    g_cost = {start: 0}

    visited = set()

    while open_list:

        # Get node with smallest heuristic value
        h, current = heapq.heappop(open_list)

        # Avoid processing a node more than once
        if current in visited:
            continue

        visited.add(current)

        # -----------------------------
        # Goal reached
        # -----------------------------

        if current == goal:

            path = []

            while current is not None:
                path.append(current)
                current = parent[current]

            path.reverse()

            return path, g_cost[goal]

        # -----------------------------
        # Check neighbors
        # -----------------------------

        for neighbor, cost in graph[current]:

            if neighbor not in visited:

                # Calculate actual path cost
                new_g = g_cost[current] + cost

                # Add neighbor using ONLY h(n)
                heapq.heappush(
                    open_list,
                    (heuristic[neighbor], neighbor)
                )

                # Store parent
                if neighbor not in parent:
                    parent[neighbor] = current
                    g_cost[neighbor] = new_g

    return None, float("inf")


# -----------------------------
# Graph
# -----------------------------

graph = {
    'A': [('B', 2), ('C', 1)],
    'B': [('D', 3)],
    'C': [('D', 1)],
    'D': [('G', 2)],
    'G': []
}


# -----------------------------
# Heuristic h(n)
# -----------------------------

heuristic = {
    'A': 5,
    'B': 4,
    'C': 3,
    'D': 2,
    'G': 0
}


# -----------------------------
# Run Greedy Best-First Search
# -----------------------------

path, total_cost = greedy_best_first(
    graph,
    'A',
    'G',
    heuristic
)

print("Greedy Best-First Path:", " -> ".join(path))
print("Total Cost:", total_cost)


# -----------------------------
# Create NetworkX graph
# -----------------------------

G = nx.DiGraph()

for node in graph:
    for neighbor, cost in graph[node]:
        G.add_edge(node, neighbor, weight=cost)


# -----------------------------
# Node positions
# -----------------------------

pos = {
    'A': (0, 1),
    'B': (1, 2),
    'C': (1, 0),
    'D': (2, 1),
    'G': (3, 1)
}


# -----------------------------
# Draw nodes
# -----------------------------

nx.draw_networkx_nodes(
    G,
    pos,
    node_color="lightblue",
    node_size=2000,
    edgecolors="black"
)

nx.draw_networkx_labels(
    G,
    pos,
    font_size=15,
    font_weight="bold"
)


# -----------------------------
# Draw all edges
# -----------------------------

nx.draw_networkx_edges(
    G,
    pos,
    edge_color="gray",
    width=2,
    arrows=True,
    arrowsize=20
)


# -----------------------------
# Display edge costs
# -----------------------------

edge_labels = nx.get_edge_attributes(
    G,
    "weight"
)

nx.draw_networkx_edge_labels(
    G,
    pos,
    edge_labels=edge_labels,
    font_size=13,
    font_weight="bold"
)


# -----------------------------
# Highlight Greedy path
# -----------------------------

path_edges = list(zip(path, path[1:]))

nx.draw_networkx_edges(
    G,
    pos,
    edgelist=path_edges,
    edge_color="red",
    width=4,
    arrows=True,
    arrowsize=25
)


# -----------------------------
# Show total cost
# -----------------------------

plt.title(
    f"Greedy Best-First Search\n"
    f"Path: {' → '.join(path)} | "
    f"Total Cost = {total_cost}",
    fontsize=16,
    fontweight="bold"
)

plt.text(
    1.5,
    -0.5,
    f"Path Cost: "
    f"{' + '.join(str(G[u][v]['weight']) for u, v in path_edges)}"
    f" = {total_cost}",
    ha="center",
    fontsize=14,
    color="red",
    fontweight="bold"
)

plt.axis("off")
plt.tight_layout()
plt.show()
